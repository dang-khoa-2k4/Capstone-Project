\section{Graph of Convex Sets (GCS)}

\begin{frame}{Ý tưởng chính (GCS)}
    \begin{itemize}
        \item \textbf{Phân hoạch không gian:} $\rightarrow$ \textbf{vùng an toàn lồi} $\{Q_v\}$
        \item \textbf{Mô hình hóa:} Hoạch định chuyển động $\rightarrow$ \textbf{SPP} trên \textbf{Graph of Convex Sets}
        \item \textbf{Quyết định:}
              \begin{itemize}
                  \item \textbf{Rời rạc:} chọn cạnh / đường đi
                  \item \textbf{Liên tục:} trạng thái / quỹ đạo trong $Q_v$
              \end{itemize}
        \item \textbf{Giải:} \textbf{relaxation} $\rightarrow$ convex optimization $\rightarrow$ đường khả thi tối ưu toàn cục
    \end{itemize}

    \begin{block}{Convex safe region}
        $Q_v \subset \mathbb{R}^n$: lồi, đóng, an toàn~\cite{marcucci2023motion}.
    \end{block}
    \centering

\end{frame}

\begin{frame}{Phân hoạch lồi chia $\mathcal{C}_{\text{free}}$ thành tập hữu hạn vùng $\{Q_v\}$ không va chạm}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.5\textwidth}

            \includegraphics[width=0.8\linewidth]{./imgs/decompose-ex-2.png}

        \begin{column}{0.5\textwidth}
            Chia $\mathcal{C}_{\text{free}}$ thành $\{Q_v\}$ sao cho
            $\bigcup_v Q_v \supseteq \mathcal{C}_{\text{free}}$.

            \vspace{0.4em}
            \textbf{Mục tiêu:}
            \begin{itemize}
                \item Số vùng tối thiểu, mỗi vùng lớn nhất có thể
                \item Phủ toàn bộ không gian tự do
            \end{itemize}

            \vspace{0.3em}
            \textbf{Ý nghĩa trong GCS:}
            \begin{itemize}
                \item Mỗi $Q_v$ $\rightarrow$ một đỉnh trong đồ thị
                \item Chất lượng phân hoạch quyết định chất lượng nghiệm
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\section{Bài toán tối ưu hóa}

\begin{frame}{Tiếp cận 1: Bài toán tối ưu hóa trong GCS}

    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \centering
            \textbf{Bài toán rời rạc}
            \[
                y_e \in \{0,1\},\quad e\in\mathcal E
            \]
            \[
                \sum_{e\in\mathcal I^{out}_v} y_e
                -
                \sum_{e\in\mathcal I^{in}_v} y_e
                =
                \begin{cases}
                    1  & v=\sigma       \\
                    -1 & v=\tau         \\
                    0  & \text{còn lại}
                \end{cases}
            \]
        \end{column}

        \begin{column}{0.02\textwidth}
            \centering
            \rule{0.4pt}{0.6\textheight}
        \end{column}

        \begin{column}{0.48\textwidth}
            \centering
            \textbf{Bài toán liên tục}
            \[
                \min_{q,T}
                aT
                + b\int_0^T \|\dot q(t)\|_2dt
                + c\int_0^T \|\dot q(t)\|_2^2dt
            \]
            \begin{itemize}
                \item $q(t)\in \bigcup_i Q_i$
                \item $\dot q(t)\in\mathcal D$
                \item $q(0)=q_{init},\;q(T)=q_{final}$
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{-0.2em}
    \[
        \Rightarrow
        \min \sum_{e=(u,v)\in\mathcal E}
        \ell_e(x_u,x_v)y_e
    \]

\end{frame}

\begin{frame}{Hoạch định chuyển động sử dụng mô hình SPP trong GCS}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.55\textwidth}
            \textbf{Mục tiêu:} tìm quỹ đạo tối ưu $\sigma \to \tau$ trên GCS
            \begin{itemize}
                \item Phân hoạch không gian thành các vùng an toàn lồi $\{Q_v \subset R^2\}$
                \item Thiết lập đồ thị liền kề GCS $G=(V,\mathcal{E})$
                \item Gán biến trạng thái $x_v \in \mathcal{X}_v$ cho mọi $v \in V$:
                      \begin{itemize}
                          \item $x_v = (q_v^{s,x}, q_v^{s,y}, q_v^{m,x}, q_v^{m,y}, q_v^{t,x}, q_v^{t,y}) \in \mathbb{R}^6$
                          \item $\mathcal{X}_v = \{x_v \mid q_v^s, q_v^m, q_v^t \in Q_v\}$
                          \item Tại $s$ và $t$: $\mathcal{X}_{\sigma} := \mathcal{X}_{\tau} := \emptyset$
                      \end{itemize}

            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \centering
            \includegraphics[width=\linewidth]{./imgs/gcs-1.png}
            \includegraphics[width=\linewidth]{./imgs/gcs-2.png}
            \captionof{figure}{GCS Framework\footnotemark}
            % \includegraphics[width=0.7\linewidth]{./imgs/gcs-3.png}

% ===================== Slide DMS-1: GCS-DMS = MINLP (overview) =====================
\begin{frame}{GCS-DMS gộp graph problem và OCP cục bộ thành một bài toán MINLP}
    \small

    \begin{columns}[T, onlytextwidth]
        \begin{column}{0.43\textwidth}
            \centering \textbf{graph problem - network flow}

            \begin{itemize}
                \item $y_{uv}\in\{0,1\}$: cạnh $(u,v)$ được chọn
                \item $p_v\in\{0,1\}$: vùng $v$ được kích hoạt
                \item Bảo toàn luồng: ${\displaystyle\sum_{\mathrm{in}}} y =
                              {\displaystyle\sum_{\mathrm{out}}} y = p_v$
                \item Interface $z_{uv}\!\in\!Q_u\!\cap\! Q_v$,\; $s_u^+ = s_v^- = z_{uv}$
            \end{itemize}

            % \vspace{0.2cm}
            \[
                J_{\mathrm{total}} = \sum_{v\in V_r} p_v\, J_v(s_v^-,w_v,\Delta_v)
            \]
        \end{column}

        \begin{column}{0.02\textwidth}
            \centering \rule{0.4pt}{0.55\textheight}
        \end{column}

        \begin{column}{0.55\textwidth}
            \centering \textbf{OCP trên vùng $v$ (khi $p_v=1$)}

            \vspace{0.2cm}
            \[
                J_v =
                \underbrace{a\Delta_v}_{\text{thời lượng}}
                +
                \underbrace{
                w_L\!\int_0^1 \tfrac{1}{\Delta_v}
                \!\left|\tfrac{dq_v}{d\tau}\right|^{\!2}\!d\tau
                }_{\text{vận tốc hình học}}
                +
                \underbrace{
                    w_E\!\int_0^1 \Delta_v|u_v|^2d\tau
                }_{\text{năng lượng điều khiển}}
            \]

            \vspace{0.1cm}
            \footnotesize
            $q_v(\tau):=\pi(x_v(\tau))$;\quad
            $\tfrac{1}{\Delta_v}\tfrac{dq_v}{d\tau}=\dot q_v$;\quad
            $\Delta_v d\tau=dt$
        \end{column}
    \end{columns}

    \vspace{0.15cm}
    \begin{alertblock}{Phân loại bài toán}
        Biến nhị phân $p_v,y_{uv}$ + OCP phi tuyến $\;\Rightarrow\;$
        \textbf{MIOCP} (liên tục theo thời gian) \;/\;
        \textbf{MINLP} (sau khi DMS rời rạc hóa điều khiển)
    \end{alertblock}

\end{frame}

% ===================== Slide DMS-2: DMS - IVP & Defect =====================
\begin{frame}{DMS tham số hóa đoạn quỹ đạo, buộc tính liên tục qua defect constraint}
    \small

    \begin{columns}[T, onlytextwidth]
        \begin{column}{0.50\textwidth}
            \textbf{Biến quyết định trên mỗi vùng $v \in V_r$:}
            \begin{itemize}
                \item $s_v^- \in \mathbb{R}^{n_x}$ - trạng thái vào (điểm bắn)
                \item $s_v^+ \in \mathbb{R}^{n_x}$ - trạng thái ra (độc lập)
                \item $\Delta_v \ge 0$ - thời lượng đoạn
                \item $w_v$ - tham số hóa điều khiển $u_v(\tau) = \mathcal{U}_v(\tau;w_v)$
            \end{itemize}

            \vspace{0.3cm}
            \textbf{IVP cục bộ} (thời gian chuẩn hóa $\tau\in[0,1]$):
            \[
                \frac{dx_v}{d\tau} = \Delta_v f\!\bigl(x_v(\tau),\,u_v(\tau)\bigr),
                \quad x_v(0) = s_v^-
            \]
        \end{column}

        \begin{column}{0.46\textwidth}
            \begin{alertblock}{Ràng buộc defect - then chốt DMS}
                \[
                    s_v^+ \;-\; \underbrace{F_v(s_v^-,\,w_v,\,\Delta_v)}_{\displaystyle x_v(1)} \;=\; 0
                \]
                Tính liên tục áp đặt \emph{sau} tích phân, không cần tích phân tuần tự.
            \end{alertblock}
        \end{column}
    \end{columns}

\end{frame}

% ===================== Slide DMS-3: Solver (2 pha) + Safety + Output =====================
\begin{frame}{Giải MINLP; ràng buộc an toàn qua lấy mẫu hoặc log-barrier}
    \footnotesize

    \begin{columns}[T, onlytextwidth]
        \begin{column}{0.52\textwidth}
            \textbf{Pha 1 - Chọn đường đi:}
            \begin{itemize}
                \item Relax $y_{uv}\in\{0,1\}$ sang $[0,1]$
                \item Giải convex problem trên toàn GCS
                \item Làm tròn để cố định chuỗi vùng $Q_{v_1}\to\cdots\to Q_{v_K}$
            \end{itemize}

            \vspace{0.15cm}
            \textbf{Pha 2 - Tối ưu quỹ đạo:}
            \begin{itemize}
                \item Warm-start từ pha 1
                \item Tối ưu đồng thời $(s_v^\pm,\,w_v,\,\Delta_v)$ trên các vùng active
            \end{itemize}

            \vspace{0.15cm}
            \textbf{Cơ chế an toàn continuous-time:}
            \begin{itemize}
                \item \textbf{Lấy mẫu:} $A_v x_v(\tau_k)\!\le\! b_v\!-\!\varepsilon_v$ -
                      đơn giản, có blind spot giữa node
                \item \textbf{Log-barrier:} $-\mu_v\!\int\!\textstyle\sum_j\!\log(b_j\!-\!a_j^\top x_v)\,d\tau$ -
                      liên tục, nhạy với $\mu_v$
            \end{itemize}
        \end{column}

        \begin{column}{0.44\textwidth}
            \textbf{Đầu ra:}
            \begin{itemize}
                \item Chuỗi vùng $Q_{v_1},\ldots,Q_{v_K}$ được kích hoạt
                \item $x^\star(\cdot)$: quỹ đạo \textbf{liên tục}, an toàn
                \item $u^\star(\cdot)$: tín hiệu điều khiển hợp lệ
                \item $T^\star = \textstyle\sum_v \Delta_v^\star$: thời gian tổng
            \end{itemize}

            \vspace{0.3cm}
            \textbf{Ba bảo đảm cùng lúc:}
            \begin{itemize}
                \item \textbf{Liên tục:} defect $= 0$ tại mọi giao điểm
                \item \textbf{An toàn:} $x^\star(t)\in Q_{v(t)},\;\forall t$
                \item \textbf{Động lực học:} IVP thỏa trên từng vùng
            \end{itemize}
        \end{column}
    \end{columns}

\end{frame}

    \begin{itemize}

        \item Phân hoạch thô $\Rightarrow$ bỏ sót vùng tự do nhỏ, giảm chất lượng nghiệm

    \begin{itemize}
        \item Vật cản di chuyển $\Rightarrow$ phải tái phân hoạch và giải lại
        \item Thời gian giải hiện tại ($>$80 s) chưa phù hợp cho real-time
    \end{itemize}

\begin{frame}{GCS vs. MICP truyền thống}
    \small
    \textbf{1. MICP Truyền thống}
    \begin{itemize}
        \item \textbf{Cơ chế:} Chia quỹ đạo thành các đoạn cố định. Với mỗi đoạn, phải gán nó vào 1 trong $|I|$ vùng an toàn thông qua biến nhị phân.
        \item \textbf{Hệ quả:} Để đảm bảo tính tổng quát (quỹ đạo có thể đi qua mọi vùng), số biến nhị phân cần là $|I| \times |I| = |I|^2$.
        \item \textbf{Thực tế:} Với 625 ô, cần \textbf{390625 biến} $\rightarrow$ Không thể giải.
    \end{itemize}

    \hrulefill

    \textbf{2. GCS: Tiếp cận dựa trên luồng đồ thị (Flow-based)}
    \begin{itemize}
        \item \textbf{Cơ chế:} Chỉ tạo biến nhị phân cho các \textbf{cạnh (edges)} nối giữa các vùng tiếp giáp nhau. Quỹ đạo được xem như một luồng chạy từ nguồn đến đích trên đồ thị.

\begin{frame}{1. Định nghĩa và Tính chất Hình học}
    \begin{block}{Đa thức Bernstein \& Đường cong Bézier}
        Đường cong Bézier bậc $d$ được xây dựng từ tập $d+1$ điểm điều khiển $\gamma_k \in \mathbb{R}^n$:
        \begin{equation*}
            \gamma(s) := \sum_{k=0}^d \beta_{k,d}(s)\gamma_k, \quad s \in [0, 1]
        \end{equation*}
        Trong đó, đa thức Bernstein là $\beta_{k,d}(s) := \binom{d}{k} s^k(1-s)^{d-k}$.
    \end{block}

    \begin{columns}
        \begin{column}{0.6\textwidth}
            \begin{itemize}
                \item \textbf{Giá trị điểm cuối:} $\gamma(0) = \gamma_0$ và $\gamma(1) = \gamma_d$.
                \item \textbf{Tính chất bao lồi:} $\gamma(s) \in \text{conv}(\{\gamma_k\}_{k=0}^d)$.
                \item \textbf{GCS Insight:} Ràng buộc an toàn liên tục $\gamma(s) \in X_v$ được đảm bảo nếu $\gamma_k \in X_v, \forall k$.
            \end{itemize}
        \end{column}

\begin{frame}{2. Đạo hàm và Tối ưu hóa trong GCS}
    \begin{itemize}
        \item \textbf{Đạo hàm (Hodograph):}
              $\dot{\gamma}(s)$ cũng là một đường cong Bézier bậc $d-1$:
              \begin{equation*}
                  \dot{\gamma}(s) = \sum_{k=0}^{d-1} \beta_{k,d-1}(s) \dot{\gamma}_k, \quad \text{với } \dot{\gamma}_k = d(\gamma_{k+1} - \gamma_k)
              \end{equation*}
              \textit{Hệ quả:} Các giới hạn vận tốc/gia tốc trở thành các ràng buộc tuyến tính trên sai phân điểm điều khiển.

              \vspace{0.3cm}
        \item \textbf{Tích phân của hàm lồi (Cost Bound):}
              Đối với hàm chi phí lồi $f$, ta có cận trên hữu hạn:
              \begin{equation*}
                  \int_0^1 f(\gamma(s))ds \le \frac{1}{d+1} \sum_{k=0}^d f(\gamma_k)
              \end{equation*}
    \end{itemize}

    % \begin{alertblock}{Kết luận ứng dụng}
    %     Cấu trúc này cho phép đưa bài toán tối ưu hóa quỹ đạo liên tục về dạng \textbf{Quy hoạch lồi hỗn hợp số nguyên (MICP)}, giúp giải quyết hiệu quả các bài toán tránh va chạm và ràng buộc động lực học.
    % \end{alertblock}
\end{frame}

\begin{frame}{Tham số hóa quỹ đạo bằng đường Bézier}
    \begin{itemize}
        \item Mỗi đoạn quỹ đạo trong $Q_i$ là đường cong Bézier bậc $m$
        \item Xác định bởi $m+1$ điểm điều khiển $\gamma_0, \ldots, \gamma_m \in Q_i$
        \item $\gamma_0 = x_i^{\text{in}}$ và $\gamma_m = x_i^{\text{out}}$
    \end{itemize}

    \begin{block}{Tính chất bao lồi của đường Bézier}
        Đường cong Bézier luôn nằm trong bao lồi của các điểm điều khiển \cite{farin2002}. Do đó, nếu tất cả điểm điều khiển thuộc miền lồi $Q_i$, toàn bộ đường cong cũng nằm trong $Q_i$ và tránh va chạm.
    \end{block}

    \begin{itemize}
        \item Có thể áp đặt ràng buộc liên tục đạo hàm tại điểm nối để quỹ đạo trơn tru
              ($C^1, C^2, \ldots$) \cite{marcucci2023motion}
    \end{itemize}
\end{frame}

\begin{frame}{Nới lỏng lồi (Convex Relaxation) và Giải thuật Làm tròn}
    \textbf{Chiến lược nới lỏng bài toán MICP:}
    \begin{itemize}

    \end{itemize}
    \textbf{Kết quả nới lỏng:}
    \begin{itemize}
        \item Kết quả cho ra thường chặt: $C_{relax} \approx C_{opt}$

        \item \textbf{Đánh giá:}

                  \item Ước lượng qua nới lỏng: $\delta_{relax} = \frac{C_{round} - C_{relax}}{C_{relax}}$
              \end{itemize}
    \end{itemize}

    \textbf{Làm tròn ngẫu nhiên:}
    \begin{enumerate}
        \item Khởi tạo từ nguồn $\sigma$
        \item Tại nút $v$: chọn cạnh $e=(v \to w)$ theo xác suất $\propto y_e^*$
        \item Quay lui nếu ngõ cụt, đến đích $\tau$~\cite{marcucci2023motion}
    \end{enumerate}

\begin{frame}{Local multiple shooting block trên mỗi vùng GCS}
    \small

    Với mỗi vùng $v\in V_r$, biến continuous cục bộ là

    \[
        s_v^- \in \mathbb{R}^{n_x},
        \qquad
        s_v^+ \in \mathbb{R}^{n_x},
        \qquad
        w_v\in\mathbb{R}^{m_v},
        \qquad
        \Delta_v\ge 0.
    \]

    Trên miền thời gian chuẩn hoá $\tau\in[0,1]$:

    \[
        \frac{dx_v}{d\tau}(\tau) =
        \Delta_v f(x_v(\tau),u_v(\tau)),
        \qquad
        x_v(0)=s_v^-,
        \qquad
        u_v(\tau)=U_v(\tau;w_v).
    \]

    Endpoint map:

    \[
        F_v(s_v^-,w_v,\Delta_v):=x_v(1).
    \]

    Defect constraint của multiple shooting:

    \[
        s_v^+ - F_v(s_v^-,w_v,\Delta_v)=0,
        \qquad
        \forall v\in V_r.
    \]

    \vspace{0.5em}

    Ý nghĩa: $s_v^-$ và $s_v^+$ là biến tối ưu độc lập, còn defect constraint buộc
    chúng khớp với nghiệm động lực học sinh ra bởi local IVP.

\end{frame}

\begin{frame}{Formulation GCS - Multiple Shooting}
    \footnotesize
    \[
        \begin{aligned}
            \min_{\substack{y,p,z,\{s_v^-,s_v^+,w_v,\Delta_v\}_{v\in V_r}}}
            \quad
             &
            \sum_{v\in V_r} p_v J_v(s_v^-,w_v,\Delta_v)
            \\
            \text{s.t.}\quad
             &
            \sum_{(\sigma,v)\in E} y_{\sigma v}=1,\quad
            \sum_{(v,\tau)\in E} y_{v\tau}=1,
            \\
             &
            \sum_{(u,v)\in E}y_{uv}
            =
            \sum_{(v,w)\in E}y_{vw}
            =
            p_v,\quad \forall v\in V_r,
            \\
             &
            s_v^+=F_v(s_v^-,w_v,\Delta_v),\quad
            x_v(\tau_{v,k})\in Q_v,\quad
            u_v(\tau_{v,k})\in\mathcal U,
            \quad \forall v,\ k,
            \\
             &
            y_{uv}=1\Rightarrow
            z_{uv}\in Q_u\cap Q_v,\quad
            s_u^+=z_{uv},\quad s_v^-=z_{uv},
            \quad \forall (u,v)\in E,
            \\
             &
            y_{\sigma v}=1\Rightarrow s_v^-=x_s,\quad
            y_{v\tau}=1\Rightarrow s_v^+=x_g,
            \\
             &
            y_{uv},p_v\in\{0,1\},\quad \Delta_v\ge0.
        \end{aligned}
    \]

    GCS chọn skeleton rời rạc $\sigma\to v_1\to\cdots\to v_M\to\tau$; multiple
    shooting biến mỗi vùng active thành một đoạn động lực học hữu hạn chiều.
